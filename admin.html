// Add this inside your index.html script when the page loads async function loadStoreSettings() { try { const doc = await db.collection('settings').doc('config').get(); if (doc.exists) { const data = doc.data(); // Update Store Name wherever it appears on the page if (data.storeName) { document.querySelectorAll('.store-name-text').forEach(el => el.innerText = data.storeName); document.title = data.storeName + " | Storefront"; } // Update Logo (supports image URL or emoji icon) if (data.logoUrl) { const logoEl = document.getElementById('store-logo'); if (logoEl) { if (data.logoUrl.startsWith('http')) { logoEl.innerHTML = ``; } else { logoEl.innerText = data.logoUrl; } } } // Update Announcement Banner Text if (data.announcementBanner) { const bannerEl = document.getElementById('announcement-banner'); if (bannerEl) bannerEl.innerText = data.announcementBanner; } // Update Footer text if (data.footerText) { const footerEl = document.getElementById('footer-text'); if (footerEl) footerEl.innerText = data.footerText; } } } catch (e) { console.error("Error loading store config:", e); } }
